                          color: active ? "#fff" : type === "positive" ? "#15803d" : "#b91c1c",
                        }}>
                          {label}
                        </span>
                      </button>
                    );
                  })}
                </div>

                {sentiment && (
                  <div style={S.tagsSection}>
                    <div style={S.tagsLabel}>What stood out?</div>
                    <div style={S.tagsWrap}>
                      {tags.map((tag) => {
                        const sel = selectedTags.includes(tag);
                        const accent = sentiment === "positive" ? "#16a34a" : "#dc2626";
                        const accentBg = sentiment === "positive" ? "#f0fdf4" : "#fff5f5";
                        const accentBorder = sentiment === "positive" ? "#bbf7d0" : "#fecaca";
                        const accentText = sentiment === "positive" ? "#15803d" : "#b91c1c";
                        return (
                          <button
                            key={tag}
                            onClick={() => toggleTag(tag)}
                            style={{
                              ...S.tag,
                              background: sel ? accent : accentBg,
                              border: `1.5px solid ${sel ? accent : accentBorder}`,
                              color: sel ? "#fff" : accentText,
                            }}
                          >
                            {sel && (
                              <svg width="10" height="10" viewBox="0 0 10 10" fill="none" style={{ marginRight: 4, flexShrink: 0 }}>
                                <path d="M1.5 5l2.5 2.5 5-5" stroke="#fff" strokeWidth="1.6" strokeLinecap="round" strokeLinejoin="round"/>
                              </svg>
                            )}
                            {tag}
                          </button>
                        );
                      })}
                    </div>
                  </div>
                )}

                <div style={S.divider} />

                <textarea
                  style={S.textarea}
                  placeholder="Anything else? Tell us what sounded off, or what you'd love the rewrite to do better."
                  value={customText}
                  onChange={(e) => setCustomText(e.target.value)}
                  rows={3}
                />

                <div style={S.contactSection}>
                  <div style={S.contactHeader}>
                    <svg width="13" height="13" viewBox="0 0 13 13" fill="none" style={{ flexShrink: 0, marginTop: 1 }}>
                      <circle cx="6.5" cy="4.5" r="2.2" stroke="#9ca3af" strokeWidth="1.2"/>
                      <path d="M1.5 11.5c0-2.485 2.239-4 5-4s5 1.515 5 4" stroke="#9ca3af" strokeWidth="1.2" strokeLinecap="round"/>
                    </svg>
                    <span>Leave your contact — we'll loop you in on updates & member perks</span>
                  </div>
                  <input
                    style={S.contactInput}
                    placeholder="Email / WeChat / Any way you prefer"
                    value={contact}
                    onChange={(e) => setContact(e.target.value)}
                  />
                  <div style={S.contactCards}>
                    <div style={S.contactCard}>
                      <div style={S.contactCardIcon}>
                        <svg width="14" height="14" viewBox="0 0 14 14" fill="none">
                          <rect x="1" y="3" width="12" height="8" rx="1.5" stroke="#4b5563" strokeWidth="1.3"/>
                          <path d="M1 5l6 3.5L13 5" stroke="#4b5563" strokeWidth="1.3" strokeLinecap="round"/>
                        </svg>
                      </div>
                      <div>
                        <div style={S.contactCardType}>Email</div>
                        <div style={S.contactCardValue}>support@verla.io</div>
                      </div>
                    </div>
                    <div style={S.contactCard}>
                      <div style={S.contactCardIcon}>
                        <svg width="14" height="14" viewBox="0 0 14 14" fill="none">
                          <path d="M6 1.5C3.515 1.5 1.5 3.3 1.5 5.5c0 1.2.6 2.27 1.55 3L2.7 11l1.9-.95c.5.14 1.02.2 1.4.2 2.485 0 4.5-1.8 4.5-4S8.485 1.5 6 1.5z" stroke="#4b5563" strokeWidth="1.3"/>
                          <path d="M8.5 6c.9-.65 3-.55 3.7.85.35.7.25 1.6-.3 2.1l.28 1.45-1.45-.72c-1.6.53-3.15-.18-3.5-1.35" stroke="#4b5563" strokeWidth="1.1" strokeLinecap="round"/>
                        </svg>
                      </div>
                      <div>
                        <div style={S.contactCardType}>WeChat</div>
                        <div style={S.contactCardValue}>Verla2edu</div>
                      </div>
                    </div>
                  </div>
                </div>

                <button
                  style={{
                    ...S.submitBtn,
                    opacity: sentiment === null ? 0.4 : 1,
                    cursor: sentiment === null ? "not-allowed" : "pointer",
                  }}
                  disabled={sentiment === null}
                  onClick={() => setSubmitted(true)}
                >
                  Submit Feedback
                </button>
              </>
            )}
          </div>
        </div>
      )}
    </div>
  );
}
